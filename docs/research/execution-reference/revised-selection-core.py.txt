def canonical(x):return json.dumps(x,ensure_ascii=False,sort_keys=True,separators=(',',':'),allow_nan=False).encode()
def sha(p):
 with Path(p).open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
def key(suite,task):return hashlib.sha256((suite+':'+task).encode()).hexdigest()[:24]
def matches(c,p):
 if 'all' in c:return all(matches(z,p) for z in c['all'])
 if 'suite' in c:return p['suite']==c['suite']
 if 'axis_kind' in c:return p['axis_kind']==c['axis_kind']
 x=p[c['field']];v=c['value']
 if x is None:return False
 return {'le':lambda:x<=v,'ge':lambda:x>=v,'gt':lambda:x>v,'eq':lambda:x==v}[c['operator']]()
def retrieve(cards,p):
 candidates=[]
 for position,c in enumerate(cards):
  applicable=matches(c['condition'],p)
  distance=sum(abs(math.log(p[f]/v)) for f,v in c.get('reference',{}).items())/len(c['reference']) if applicable and 'reference' in c else 0 if applicable else None
  candidates.append(dict(card_id=c['id'],layer=c['layer'],applicable=applicable,distance=distance,tie_order=position))
 chosen=[]
 for layer,cap in [('数据类型',1),('数理特征',2),('结构与质量',1)]:
  group=sorted((x for x in candidates if x['layer']==layer and x['applicable']),key=lambda x:(x['distance'],x['tie_order']))[:cap]
  chosen.extend(group)
 ids=[x['card_id'] for x in chosen]
 return [dict(id=c['id'],layer=c['layer'],title=c['title'],text=c['text']) for i in ids for c in cards if c['id']==i],dict(candidates=candidates,selected=chosen,rule='condition match then mean absolute natural-log proportional distance to reviewed Train-profile median; fixed card order breaks ties')

 candidates=[]
 for model in MODELS:
  fact=row['models'][model];fixed=fact['fixed_config']
  params={k:v for k,v in fixed.items() if k not in ['model_type','normalization','class_balance','seed','feature_selection_enabled','split_mode','split_train','split_valid','split_test']}
  item=dict(model_id=model,display_name=NAMES[model],model_params=params)
  if model=='cnn1d':item['train_resolved_model_profile']=read_json(Path(fact['artifact_directory'])/'config.json')['model_profile']

  shared_conditions=dict(normalization='zscore',class_balance='none',feature_selection_enabled=False,seed=42,max_parameter_candidates=1,classical_final_fit='original Train plus Valid refit of the frozen default',deep_final_fit='best original Valid-loss checkpoint; no refit'),candidates=candidates,retrieved_knowledge=knowledge)

  elif isinstance(obj,list):

 check(semantic)

def parse_content(text):
 raw_valid=False
 try:_strict_json(text);raw_valid=True
 except ValueError:pass
 trace={};action=_semantic_action(text,trace)
 operations=trace.get('semantic_format_normalization',{}).get('operations',[])
 if set(operations)-{'append_final_object_brace','trim_rationale_edge_whitespace'}:raise ValueError('Only the reviewed trailing outer brace repair is permitted')
 if type(action) is not dict or set(action)!={'tool_name','arguments','rationale'}:raise ValueError('Invalid action fields')
 if action['tool_name']!='submit_ml_experiment' or type(action['arguments']) is not dict or set(action['arguments'])!={'model_id'}:raise ValueError('Invalid tool or arguments')
 if type(action['arguments']['model_id']) is not str or action['arguments']['model_id'] not in MODELS:raise ValueError('Model is not in the frozen seven-model menu')
 rationale=_rationale(action['rationale'])
 return dict(model_id=action['arguments']['model_id'],rationale=rationale,raw_json_valid=raw_valid,format_operations=operations,normalized_action=action,normalization_trace=trace)
